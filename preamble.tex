% so this renders identically on any machine with no extra setup.

% --- Colours ---------------------------------------------------
\definecolor{ink}{HTML}{1A1815}
\definecolor{muted}{HTML}{6E6862}

\color{ink}

% --- Measurements ----------------------------------------------
% Width reserved in the left margin for section headings. The rail sits
% inside the page's left margin, so `left` in the geometry block of cv.qmd
% must be this plus the outer edge, or the rail runs off the paper.
\newlength{\marginheadwidth}
\setlength{\marginheadwidth}{1.35in}
% Gap between the heading rail and the text column.
\newlength{\margingutter}
\setlength{\margingutter}{0.25in}
% Width reserved for dates in dated entries.
\newlength{\datewidth}
\setlength{\datewidth}{\dimexpr\marginheadwidth-\margingutter\relax}

% --- Section headings sit in the left margin -------------------
\titleformat{\section}[leftmargin]
  {\normalfont\sffamily\bfseries\small\color{ink}\raggedleft}
  {}{0pt}{}
\titlespacing*{\section}{\marginheadwidth}{1.6em}{\margingutter}

\titleformat{\subsection}[hang]
  {\normalfont\sffamily\small\color{muted}}
  {}{0pt}{}
\titlespacing*{\subsection}{0pt}{0.9em}{0.25em}

% Keep a heading from stranding itself at the foot of a page.
\pretocmd{\section}{\needspace{3\baselineskip}}{}{}
\pretocmd{\subsection}{\needspace{3\baselineskip}}{}{}

% --- Lists -----------------------------------------------------
% Dated entries: the date hangs in its own fixed-width column.
\renewcommand{\descriptionlabel}[1]{%
  \llap{\makebox[\datewidth][r]{\normalfont\sffamily\footnotesize\color{muted}#1}%
        \hspace{\margingutter}}}
\setlist[description]{
  leftmargin=0pt,
  labelsep=0pt,
  itemindent=0pt,
  itemsep=0.5em,
  parsep=0pt,
  topsep=0.35em
}
\setlist[itemize]{leftmargin=1.1em, itemsep=0.25em, topsep=0.3em, parsep=0pt, label=\textcolor{rulecol}{\rule[0.25ex]{0.4em}{0.5pt}}}
\setlist[enumerate]{leftmargin=1.8em, itemsep=0.35em, topsep=0.3em, parsep=0pt, font=\sffamily\footnotesize\color{muted}}
